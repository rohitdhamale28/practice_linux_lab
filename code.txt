app.py
from flask import Flask

app = Flask(__name__)

@app.route("/")
def hello():
    return "Hello World from Flask in Docker!"

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)



FROM    python:3.11-slim

WORKDIR     /app

COPY     app.py     .

RUN    pip     install     flask

EXPOSE     5000

CMD     ["python",  "app.py"]
docker run -it -p 5050:5000 flaskp


docker run -d -p 9000:80 --name nginxp nginx:latest
